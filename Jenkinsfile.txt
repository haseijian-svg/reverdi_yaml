pipeline {
    agent any
    environment {
        REGISTRY = '10.0.50.10:5000'
        IMAGE_NAME = 'reverdi-app'
        GITOPS_REPO = 'https://github.com/epqlffltm/reluxe-gitops.git'
    }
    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }
        stage('Build & Test') {
            steps {
                sh 'docker build -t ${REGISTRY}/${IMAGE_NAME}:${BUILD_NUMBER} .'
            }
        }
        stage('Push to Registry') {
            steps {
                sh 'docker push ${REGISTRY}/${IMAGE_NAME}:${BUILD_NUMBER}'
            }
        }
        stage('Update GitOps Tag') {
            steps {
                withCredentials([usernamePassword(credentialsId: 'github-token', passwordVariable: 'GIT_TOKEN', usernameVariable: 'GIT_USER')]) {
                    sh '''
                        git clone https://${GIT_USER}:${GIT_TOKEN}@github.com/epqlffltm/reluxe-gitops.git gitops-repo
                        cd gitops-repo
                        sed -i "s/tag:.*/tag: \\"${BUILD_NUMBER}\\"/g" charts/reverdi-app/values-vagrant.yaml
                        git config user.name "jenkins-bot"
                        git config user.email "jenkins@reverdi.internal"
                        git commit -am "chore(release): bump image tag to ${BUILD_NUMBER}"
                        git push origin main
                    '''
                }
            }
        }
    }
}
